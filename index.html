async function fetchAndRenderWarnings() { try { const response = await fetch('https://www.jma.go.jp/bosai/warning/data/r8/280000.json'); if (!response.ok) throw new Error('Network response failed'); const data = await response.json(); const hyogoWarnings = new Set(); const hyogoAdvisories = new Set(); const kakogawaWarnings = []; const kakogawaAdvisories = []; // コードから名称への変換用（主要なもの） const getWarningName = (code) => { const names = { "03": "大雨警報", "04": "洪水警報", "05": "暴風警報", "06": "大雪警報", "07": "波浪警報", "08": "高潮警報", "10": "大雨注意報", "12": "強風注意報", "19": "雷注意報", "21": "濃霧注意報", "22": "乾燥注意報" }; return names[code] || `コード(${code})`; }; const isWarning = (code) => parseInt(code, 10) >= 2 && parseInt(code, 10) <= 9; // JSON内を走査 data.forEach(report => { if (report.warning && report.warning.class20Items) { report.warning.class20Items.forEach(item => { if (item.kinds) { item.kinds.forEach(k => { if (k.status === '発表' || k.status === '継続') { const name = getWarningName(k.code); // 1. 兵庫県全体の集約用プール if (isWarning(k.code)) { hyogoWarnings.add(name); } else { hyogoAdvisories.add(name); } // 2. 加古川市（2821000）の個別抽出用プール if (item.areaCode === '2821000') { if (isWarning(k.code)) { kakogawaWarnings.push(name); } else { kakogawaAdvisories.push(name); } } } }); } }); } }); // 兵庫県全体の表示を構築 const hyogoEl = document.getElementById('hyogo-warning-container'); if (hyogoWarnings.size > 0 || hyogoAdvisories.size > 0) { hyogoEl.innerHTML = ` ${hyogoWarnings.size > 0 ? `
⚠️ 警報: ${[...hyogoWarnings].join('、')}
` : ''} ${hyogoAdvisories.size > 0 ? `
注意報: ${[...hyogoAdvisories].join('、')}
` : ''} `; } else { hyogoEl.innerHTML = `
現在、兵庫県内に発表されている警報・注意報はありません。
`; } // 加古川市の表示を構築 const kakogawaEl = document.getElementById('kakogawa-warning-container'); if (kakogawaWarnings.length > 0 || kakogawaAdvisories.length > 0) { kakogawaEl.innerHTML = ` ${kakogawaWarnings.length > 0 ? `
⚠️ 警報: ${[...new Set(kakogawaWarnings)].join('、')}
` : ''} ${kakogawaAdvisories.length > 0 ? `
注意報: ${[...new Set(kakogawaAdvisories)].join('、')}
` : ''} `; } else { kakogawaEl.innerHTML = `
現在、加古川市に発表されている警報・注意報はありません。
`; } } catch (e) { console.error(e); } }
